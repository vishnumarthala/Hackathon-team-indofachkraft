const jars: Record<string, string> = {};
const cookieJar: Record<string, { name: string; value: string }[]> = {};
async function signIn(key: string) {
  const store = new Map<string, string>();
  const client = createServerClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!, {
    cookies: { getAll: () => [...store].map(([name, value]) => ({ name, value })), setAll: (list) => list.forEach((c) => store.set(c.name, c.value)) },
  });
  const { error } = await client.auth.signInWithPassword({ email: users[key].email, password: users[key].password });
  if (error) throw new Error(`sign in ${key}: ${error.message}`);
  cookieJar[key] = [...store].map(([name, value]) => ({ name, value }));
  jars[key] = [...store].map(([n, v]) => `${n}=${v}`).join("; ");
}

async function api(who: string | null, method: string, path: string, body?: unknown) {
  const res = await fetch(`${BASE}${path}`, {
    method,
    headers: { ...(who ? { cookie: jars[who] } : {}), ...(body === undefined ? {} : { "content-type": "application/json" }) },
    body: body === undefined ? undefined : JSON.stringify(body),
    redirect: "manual",
  });
  const text = await res.text();
  let json: Record<string, unknown> = {};
  try { json = JSON.parse(text); } catch { /* html */ }
  return { status: res.status, json, text };
}

// ---------- setup: providers, services, children ----------
const IST = 330 * 60_000;
function slot(hoursAhead: number, extra = 0): string {
  const step = 30 * 60_000;
  const local = Date.now() + hoursAhead * 3_600_000 + IST;
  const day = Math.floor(local / 86_400_000) * 86_400_000;
  return new Date(day + Math.ceil((local - day) / step) * step + extra * step - IST).toISOString();
}

async function profile(userKey: string, orgId: string, slug: string, name: string) {
  const [loc] = await sql<{ id: string }[]>`
    insert into practice_locations (org_id, name, address_line1, locality, pincode, lat, lng)
    values (${orgId}, ${`${name} clinic (synthetic)`}, 'Road 2 (synthetic)', 'Banjara Hills', '500034', 17.412, 78.441) returning id`;
  const [p] = await sql<{ id: string }[]>`
    insert into provider_profiles (user_id, slug, display_name, provider_type, verification_status, verified_at, listed, claimed, is_synthetic)
    values (${users[userKey].id}, ${slug}, ${name}, 'speech_language_pathologist', 'verified', now(), true, true, false) returning id`;
  await sql`insert into provider_locations (provider_profile_id, location_id, org_id) values (${p.id}, ${loc.id}, ${orgId})`;
  for (const weekday of [0, 1, 2, 3, 4, 5, 6]) {
    await sql`insert into availability_rules (provider_profile_id, org_id, location_id, weekday, start_time, end_time, mode, valid_from)
              values (${p.id}, ${orgId}, ${loc.id}, ${weekday}, '00:00', '24:00', 'both', current_date - 2)`;
  }
  const [svc] = await sql<{ id: string }[]>`
    insert into services (provider_profile_id, org_id, location_id, name, mode, duration_min, fee_inr, payment_mode)
    values (${p.id}, ${orgId}, ${loc.id}, 'Speech session (synthetic)', 'in_clinic', 30, 0, 'free') returning id`;
  return { profileId: p.id, serviceId: svc.id, slug };
}

async function child(guardianKey: string, name: string) {
  const [c] = await sql<{ id: string }[]>`
    insert into children (display_name, date_of_birth, source, created_by) values (${name}, '2021-05-01', 'guardian', ${users[guardianKey].id}) returning id`;
  await sql`insert into guardian_links (child_id, user_id, relationship, status, verification_method, declared_at)
            values (${c.id}, ${users[guardianKey].id}, 'mother', 'active', 'self_registered', now())`;
  return c.id;
}

let n = 0;
async function book(guardian: string, childId: string, svc = A.serviceId) {
  const h = await api(guardian, "POST", "/api/appointments/hold", { service_id: svc, start: slot(4, n++ * 2), idempotency_key: `e036-${tag}-${n}` });
  if (h.status !== 201) throw new Error(`hold ${h.status} ${h.text}`);
  const id = h.json.appointment_id as string;
  const c = await api(guardian, "POST", `/api/appointments/${id}/confirm`, { child_id: childId });
  if (c.status !== 200) throw new Error(`confirm ${c.status} ${c.text}`);
  return id;
}
async function travel(id: string) {

  await sql`update appointments set slot = tstzrange(now() - make_interval(hours => ${n + 3}), now() - make_interval(hours => ${n + 3}) + interval '30 minutes', '[)') where id = ${id}`;
}
